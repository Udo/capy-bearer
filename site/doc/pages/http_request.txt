:title
http_request

:sig
DValue http_request(DValue req)

:content
`http_request()` sends an outbound HTTP or HTTPS request from a `dval` request map. It returns a `dval` map with `status`, `headers`, `body`, and `error`.

Set `method`, `url`, `headers`, `body`, `timeout_ms`, and `follow_redirects` as needed. Use it for an outbound request when the request handler can wait for the response.

For hardened HTTPS requests, set every `security` field to `true`: `https_only`, `public_dns_only`, `pin_dns`, `isolated_curl`, and `no_redirects`. A partial or false hardened policy returns `invalid_request`.

The hardened policy rejects non-HTTPS URLs, IP hosts, URL userinfo, unsafe DNS results, and redirects. It limits request and response bodies to 64 KiB and response headers to 8 KiB.

The `error` value can report `invalid_request`, `unsafe_dns`, `timeout`, `response_too_large`, `redirect_not_allowed`, `http_status`, or `network_failure`. In C++ `.uce`, use `DValue` for the request and response.

:see
>socket

:example capy cli
var request := dval({"method": "GET", "url": "http://127.0.0.1/doc/", "headers": {"Host": "uce.openfu.com"}, "timeout_ms": 2000})
var response := http_request(request)
print("HTTP ", s32(response["status"]), ", ", length(string(response["body"])), " bytes returned\n")

:example cpp cli
DValue req; req["method"] = "GET"; req["url"] = "http://127.0.0.1/doc/";
req["headers"]["Host"] = "uce.openfu.com"; req["timeout_ms"] = (f64)2000;
DValue resp = http_request(req);
print("HTTP ", resp["status"].to_u64(), ", ", resp["body"].to_string().length(), " bytes returned\n");


:example capy cli
// GitHub token exchange: client_secret stays in stdin body, not argv.
var client_id := "example-client"
var client_secret := "example-secret"
var code := "example-code"
var token := dval({"method": "POST", "url": "https://github.com/login/oauth/access_token", "headers": {"Accept": "application/json", "Content-Type": "application/x-www-form-urlencoded"}, "body": "client_id=" + uri_encode(client_id) + "&client_secret=" + uri_encode(client_secret) + "&code=" + uri_encode(code), "security": {"https_only": true, "public_dns_only": true, "pin_dns": true, "isolated_curl": true, "no_redirects": true}})
var token_response := http_request(token)
print("hardened token exchange composed; status=", s32(token_response["status"]), " error=", token_response["error"], "\n")

:example cpp cli
// GitHub token exchange: client_secret stays in stdin body, not argv.
String client_id="example-client", client_secret="example-secret", code="example-code";
DValue token; token["method"]="POST"; token["url"]="https://github.com/login/oauth/access_token";
token["headers"]["Accept"]="application/json"; token["headers"]["Content-Type"]="application/x-www-form-urlencoded";
token["body"]="client_id="+uri_encode(client_id)+"&client_secret="+uri_encode(client_secret)+"&code="+uri_encode(code);
for(String key:{"https_only","public_dns_only","pin_dns","isolated_curl","no_redirects"}) token["security"][key].set_bool(true);
DValue token_response=http_request(token);
print("hardened token exchange composed; status=", token_response["status"].to_u64(), " error=", token_response["error"].to_string(), "\n");
