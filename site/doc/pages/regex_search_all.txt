:sig
function regex_search_all(pattern : string, subject : string, flags : string = "") dval

:params
pattern : regular expression
subject : text to search
flags : regular expression flags

:returns
Data for all non-overlapping matches.

:content
The result contains `matched`, `count`, and `matches`.

:example capy render
var matches := regex_search_all("[a-z]+", "capy docs")
print(s32(matches.count), "\n")
