:sig
bool path_is_within(String path, String root)

:params
path : the path to test
root : the directory that must contain the path

:returns
`true` if `path` is inside `root`

:content
`path_is_within()` returns `true` when `path` is inside `root`. It resolves both paths with `path_real()`. Both paths must exist. Resolution follows symbolic links and removes `.` and `..` segments.

Use this check before you open a user-supplied path.

:example capy render
var root := file_temp("/tmp/doc-within-")
file_unlink(root)
mkdir(root)
var path := path_join(root, "app.txt")
file_put_contents(path, "x")
if path_is_within(path, root) { print("inside / ") } else { print("outside / ") }
if path_is_within(path_join(root, "../" + basename(root) + "/app.txt"), root) { print("inside\n") } else { print("outside\n") }
dir_remove(root, true)

:example cpp render
mkdir("/tmp/doc-within");
file_put_contents("/tmp/doc-within/app.txt", "x");
print(path_is_within("/tmp/doc-within/app.txt", "/tmp/doc-within") ? "inside" : "outside", " / ");
print(path_is_within("/tmp/doc-within/../doc-within/app.txt", "/tmp/doc-within") ? "inside" : "outside", "\n");
dir_remove("/tmp/doc-within", true);
